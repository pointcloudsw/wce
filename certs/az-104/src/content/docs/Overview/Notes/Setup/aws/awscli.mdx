### [Source](https://docs.aws.amazon.com/cli/latest/userguide/cli-configure-sso.html#cli-configure-sso-configure)

```sh
cd ~
sudo apt update
sudo apt install unzip
curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip
sudo ./aws/install
```

#### Create AWS SSO profile 
```sh
SSOACCT=710271911790
DEFPROF=sbpwr
PROFREG=us-east-2
PROFOUT=json
SSOSESS=ssosess
SSOREG=us-east-1
SSOROLE=PowerUserAccess
SSOSTARTURL=https://d-9067de7a6e.awsapps.com/start

cd ~
mkdir .aws
cat >> config <<-EOF
    [profile ${DEFPROF}]
    sso_account_id=${SSOACCT}
    sso_role_name=${SSOROLE}
    sso_session=${SSOSESS}
    region=${PROFREG}
    output=${PROFOUT}

    [sso-session ${SSOSESS}]
    sso_region=${SSOREG}
    sso_registration_scopes=sso:account:access
    sso_start_url=${SSOSTARTURL}
EOF
```

#### SSO login via AWS CLI
```sh
aws sso login --profile sbpwr --no-browser --use-device-code

# Verify
aws sts get-caller-identity --profile sbpwr
```
